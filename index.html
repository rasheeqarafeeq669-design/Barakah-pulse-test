function process() { ctx.drawImage(v, 0, 0, 40, 40); const data = ctx.getImageData(0, 0, 40, 40).data; let r = 0, g = 0; for (let i = 0; i < data.length; i += 16) { r += data[i]; // Red channel g += data[i + 1]; // Green channel } // FINGER DETECTION: വിരൽ വെച്ചാൽ റെഡ് ചാനൽ ഗ്രീനിനേക്കാൾ വളരെ കൂടുതലായിരിക്കും if (r > g * 2) { frames.push({ v: r, t: Date.now() }); statusDisplay.innerText = "Measuring... Keep steady."; } else { frames = []; // വിരൽ മാറ്റിയിട്ടുണ്ടെങ്കിൽ പഴയ റീഡിംഗ് കളയുന്നു bpmDisplay.innerText = "--"; stressDisplay.innerText = "0%"; statusDisplay.innerText = "Place your finger firmly on camera & flash."; } if (frames.length > 90) frames.shift(); if (frames.length > 60) { let peaks = 0; for (let i = 1; i < frames.length - 1; i++) { if (frames[i].v > frames[i-1].v && frames[i].v > frames[i+1].v) peaks++; } let duration = (frames[frames.length-1].t - frames[0].t) / 60000; let bpm = Math.round(peaks / duration / 2); if (bpm > 45 && bpm < 155) { bpmDisplay.innerText = bpm; let stress = Math.min(Math.max(bpm - 55, 5), 98); stressDisplay.innerText = stress + "%"; } } setTimeout(() => requestAnimationFrame(process), 50); }
